const failure = (error: unknown) => {
  const e = error as { code?: unknown; ext?: string[]; message: string };
  const code = Array.isArray(e.ext) ? e.ext[0].replace('walleterm:', '') : typeof e.code === 'string' ? e.code : 'internal';

function call(children: xdr.SorobanAuthorizedInvocation[] = [], arg = xdr.ScVal.scvU32(42)) {
  return new xdr.SorobanAuthorizedInvocation({
    function: xdr.SorobanAuthorizedFunction.sorobanAuthorizedFunctionTypeContractFn(
      new xdr.InvokeContractArgs({
        contractAddress: new Address(contract(2)).toScAddress(),
        functionName: 'arbitrary_call',
        args: [arg, xdr.ScVal.scvSymbol('reviewed')],
      }),
    ),
    subInvocations: children,
  });
}
const chain = (levels: number) => {
  let node = call();
  for (let i = 1; i < levels; i++) node = call([node]);
  return node;
};
const wide = (contexts: number) => call(Array.from({ length: contexts - 1 }, () => call()));

function entry(opts: {
  address?: string;
  invocation?: xdr.SorobanAuthorizedInvocation;
  nonce?: bigint;
  expiration?: number;
  credentials?: 'v2' | 'v1' | 'source';
  signature?: xdr.ScVal;
}) {
  const address = opts.address ?? G;
  const credentials = new xdr.SorobanAddressCredentials({
    address: new Address(address).toScAddress(),
    nonce: opts.nonce ?? -123n,
    signatureExpirationLedger: opts.expiration ?? 160,
    signature: opts.signature ?? xdr.ScVal.scvVoid(),
  });
  const c =
    opts.credentials === 'v1'
      ? xdr.SorobanCredentials.sorobanCredentialsAddress(credentials)
      : opts.credentials === 'source'
        ? xdr.SorobanCredentials.sorobanCredentialsSourceAccount()
        : xdr.SorobanCredentials.sorobanCredentialsAddressV2(credentials);
  return new xdr.SorobanAuthorizationEntry({ rootInvocation: opts.invocation ?? call([call()]), credentials: c }).toXDR(
    'base64',
  );
}

function auth(id: string, input: Record<string, unknown>, latest: unknown = 100) {
  const request = input as never;
  let expect;
  try {
    const checked = inspectAuthEntry(request, G, latest as number);
    const signature = hexOf(key.sign(Buffer.from(checked.digest)));
    const signed = attachAuthSignature(request, G, latest as number, signature);
    expect = { digest: checked.details.hash, signature, signed_auth_entry_xdr: signed };
  } catch (error) {
    expect = failure(error);
  }
  cases.push({ id, kind: 'auth_entry', input, latest_ledger: latest, expect });
}

function preimageXdr(opts: {
  address?: string;
  expiration?: number;
  network?: string;
  legacy?: boolean;
  invocation?: xdr.SorobanAuthorizedInvocation;
}) {
  const common = {
    networkId: hash(Buffer.from(opts.network ?? Networks.TESTNET)),
    nonce: 55n,
    signatureExpirationLedger: opts.expiration ?? 220,
    invocation: opts.invocation ?? call([call()]),
  };
  return (
    opts.legacy
      ? xdr.HashIdPreimage.envelopeTypeSorobanAuthorization(new xdr.HashIdPreimageSorobanAuthorization(common))
      : xdr.HashIdPreimage.envelopeTypeSorobanAuthorizationWithAddress(
          new xdr.HashIdPreimageSorobanAuthorizationWithAddress({
            ...common,
            address: new Address(opts.address ?? G).toScAddress(),
          }),
        )
  ).toXDR('base64');
}
function preimage(id: string, preimage_xdr: string, latest: number | null = 100, network = Networks.TESTNET, public_key = G) {
  let expect;
  try {
    const checked = inspectAuthPreimage(preimage_xdr, public_key, network, latest ?? undefined);
    expect = { digest: checked.details.hash, address: checked.details.address } as Record<string, string>;
    if (latest !== null && network === Networks.TESTNET) {
      const signature = hexOf(key.sign(Buffer.from(checked.digest)));
      const input = { kind: 'auth_entry', preimage_xdr, network_passphrase: network, address: public_key } as never;
      expect.signature = signature;
      expect.signed_auth_entry = attachPreimageSignature(input, public_key, latest, signature);
    }
  } catch (error) {
    expect = failure(error);
  }
  cases.push({ id, kind: 'preimage', preimage_xdr, public_key, network_passphrase: network, latest_ledger: latest, expect });
}

const NOW_MS = 1_800_000_000_000;
const NOW = NOW_MS / 1000;
function build(opts: {
  source?: string;
  opSource?: string;
  minTime?: number;
  maxTime?: number;
  noBounds?: boolean;
  signatures?: number;
  signedBy?: Keypair;
  ops?: xdr.Operation[];
  fee?: string;
  soroban?: boolean;
}) {
  const src = opts.source ?? G;
  const account = StrKey.isValidMed25519PublicKey(src) ? new MuxedAccount(new Account(MuxedAccount.fromAddress(src, '0').baseAccount().accountId(), '41'), '7') : new Account(src, '41');
  const builder = new TransactionBuilder(account as never, { fee: opts.fee ?? '250', networkPassphrase: Networks.TESTNET });
  const ops = opts.ops ?? [Operation.payment({ destination: OTHER, asset: Asset.native(), amount: '1', source: opts.opSource })];
  for (const op of ops) builder.addOperation(op);
  if (opts.noBounds) builder.setTimeout(0);
  else builder.setTimebounds(opts.minTime ?? NOW - 10, opts.maxTime ?? NOW + 120);
  if (opts.soroban) builder.setSorobanData(new SorobanDataBuilder().build());
  const tx = builder.build();
  if (opts.signedBy) tx.sign(opts.signedBy);
  for (let i = 0; i < (opts.signatures ?? 0); i++) tx.sign(Keypair.fromRawEd25519Seed(Buffer.alloc(32, 100 + i)));
  return tx;
}
function transaction(id: string, txXdr: string, over: { network?: string; address?: string; now?: number } = {}) {
  const input = { kind: 'transaction', xdr: txXdr, network_passphrase: over.network ?? Networks.TESTNET, address: over.address ?? G } as never;
  const now = over.now ?? NOW_MS;
  let expect;
  try {
    const { details, expires } = inspectTransaction(input, G, now);
    const signature = hexOf(key.sign(Buffer.from(details.hash, 'hex')));
    expect = { hash: details.hash, details, expires_ms: expires, signature, signed_tx_xdr: attachSignature(input, G, signature) };
  } catch (error) {
    expect = failure(error);
  }
  cases.push({ id, kind: 'transaction', input, now_ms: now, expect });
}
