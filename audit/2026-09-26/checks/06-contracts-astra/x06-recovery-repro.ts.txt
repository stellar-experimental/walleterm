// Execute the frozen x06 and stepper bodies with offline state and read stubs.
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { createHash } from 'node:crypto';

const source = readFileSync('/private/tmp/walleterm-audit-40d6cca9db73/tests/cap85.ts', 'utf8');
const stepper = source.slice(source.indexOf('const stepper ='), source.indexOf('// The saved hash of an uploaded artifact.'));
const x06 = source.slice(source.indexOf('async function x06('), source.indexOf('// Observation only:'));
assert.ok(stepper.startsWith('const stepper ='));
assert.ok(x06.startsWith('async function x06('));
const executable = new Bun.Transpiler({ loader: 'ts' }).transformSync(`${stepper}\n${x06}`);

async function check(finalStepSaved: boolean) {
  let chain = finalStepSaved ? { type: 'wasm', version: 1 } : { type: 'external_ref', version: 2 };
  let writes = 0;
  let operations = 0;
  const records: string[] = [];
  const state = {
    contracts: { manager: { id: 'mock-manager' } },
    steps: { 'X06:adopt-ref': { hash: 'mock-ref-hash' } } as Record<string, object>,
  };
  if (finalStepSaved) state.steps['X06:adopt-wasm'] = { hash: 'mock-wasm-hash' };
  const context = { sdk: {}, keys: { b: { publicKey: 'mock-admin' } }, record: (id: string) => records.push(id) };
  const dependencies = {
    reconcileInflight: async () => {},
    saveState: () => { writes++; },
    deployWasm: async () => 'mock-plain',
    addr: (_sdk: unknown, value: string) => value,
    str: (_sdk: unknown, value: string) => value,
    bytes: (_sdk: unknown, value: Uint8Array) => value,
    readInstanceExecutable: async () => ({ type: chain.type }),
    readU32: async () => chain.version,
    readOptionalHash: async () => 'mock-resolved-hash',
    gAuthorizer: () => 'mock-authorizer',
    call: (_sdk: unknown, _contract: string, method: string) => method,
    wasmHash: () => 'ab'.repeat(32),
    invokeOperation: async (_ctx: unknown, _state: unknown, options: { operation: string }) => {
      assert.equal(options.operation, 'adopt_wasm');
      operations++;
      chain = { type: 'wasm', version: 1 };
      return { hash: 'mock-wasm-hash' };
    },
    FILES: { v1: 'mock-v1.wasm' }, TAG: 'target',
  };
  const run = new Function(...Object.keys(dependencies), `${executable}; return x06;`)(...Object.values(dependencies));
  let error: string | undefined;
  try { await run(context, state, {}); } catch (value) { error = String(value); }
  if (finalStepSaved) {
    assert.match(error ?? '', /adopt_ref did not move the contract onto the v2 reference/);
    assert.equal(operations, 0);
    assert.equal(writes, 0);
  } else {
    assert.equal(error, undefined);
    assert.equal(operations, 1);
  }
  return { final_step_saved: finalStepSaved, error, operations, writes, records };
}

console.log(JSON.stringify({
  status: 'passed', defect_reproduced: true,
  frozen_source_sha256: createHash('sha256').update(source).digest('hex'),
  control: await check(false), interrupted_after_final_step: await check(true),
  limits: 'Exact x06 and stepper bodies; mocked deployment, reads, persistence, and operation execution. No keys or network.',
}, null, 2));
