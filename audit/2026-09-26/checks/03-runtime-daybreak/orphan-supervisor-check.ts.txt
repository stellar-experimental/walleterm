import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { setTimeout as delay } from 'node:timers/promises';

interface ChildRecord {
  supervisor_pid: number;
  cloudflared_pid: number;
}

function alive(pid: number): boolean {
  try {
    process.kill(pid, 0);
    return true;
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === 'ESRCH') return false;
    throw error;
  }
}

const directory = mkdtempSync(join(tmpdir(), 'walleterm-supervisor-crash-'));
const frozen = '/private/tmp/walleterm-audit-40d6cca9db73';
let record: ChildRecord | undefined;

try {
  writeFileSync(
    join(directory, 'cloudflared'),
    `#!${process.execPath}\nprocess.on('SIGTERM',()=>{}); setInterval(()=>{},1000); console.log('ready');\n`,
    { mode: 0o700 },
  );
  const supervisor = spawn(process.execPath, [join(frozen, 'bridge/tunnel-child.ts')], {
    cwd: directory,
    env: { PATH: `${directory}:${process.env.PATH ?? ''}` },
    stdio: ['pipe', 'pipe', 'pipe'],
  });
  await new Promise<void>((resolve, reject) => {
    supervisor.stdout.once('data', () => resolve());
    supervisor.once('error', reject);
    supervisor.once('exit', (code, signal) => reject(Error(`early exit ${code ?? signal}`)));
  });
  record = JSON.parse(readFileSync(join(directory, 'child.json'), 'utf8')) as ChildRecord;
  assert.equal(record.supervisor_pid, supervisor.pid);
  supervisor.kill('SIGKILL');
  await new Promise<void>((resolve) => supervisor.once('exit', () => resolve()));
  await delay(100);
  const result = {
    supervisor_alive: alive(record.supervisor_pid),
    cloudflared_alive_after_supervisor_sigkill: alive(record.cloudflared_pid),
  };
  console.log(JSON.stringify(result));
  assert.equal(result.supervisor_alive, false);
  assert.equal(result.cloudflared_alive_after_supervisor_sigkill, true);
} finally {
  if (record?.cloudflared_pid && alive(record.cloudflared_pid))
    process.kill(record.cloudflared_pid, 'SIGKILL');
  rmSync(directory, { recursive: true, force: true });
}
