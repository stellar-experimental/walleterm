import { test, onTestFinished } from 'bun:test';
import assert from 'node:assert/strict';
import { setTimeout as delay } from 'node:timers/promises';
import { Account, Asset, Keypair, Networks, Operation, TransactionBuilder } from '@stellar/stellar-sdk';
import { createBridge, validOrigin } from '/private/tmp/walleterm-audit-40d6cca9db73/bridge/server.ts';
import { inspectTransaction, attachSignature } from '/private/tmp/walleterm-audit-40d6cca9db73/bridge/transaction.ts';
import type { BridgeOptions } from '/private/tmp/walleterm-audit-40d6cca9db73/bridge/server.ts';

// Isolated offline keys only. No fixture calls the real signer or 1Password.
const key = Keypair.random(), other = Keypair.random();
const address = key.publicKey();
const website = 'https://audit.example';
function builder() {
  return new TransactionBuilder(new Account(address, '5'), { fee: '100', networkPassphrase: Networks.TESTNET })
    .addOperation(Operation.manageData({ name: 'audit', value: 'offline' }));
}
function input(id = 'audit') {
  return { id, public_key: address, network_passphrase: Networks.TESTNET, transaction_xdr: builder().setTimeout(180).build().toXDR() };
}
async function until(check: () => Promise<boolean> | boolean) {
  for (let i = 0; i < 400; i++) { if (await check()) return; await delay(5); }
  throw Error('The offline fixture timed out.');
}
async function fixture(options: BridgeOptions = {}) {
  let signs = 0;
  const bridge = createBridge({ port: 0, log: () => {},
    listSigners: async () => [{ public_key: address }],
    sign: async (_publicKey, hash) => { signs++; return Buffer.from(key.sign(Buffer.from(hash, 'hex'))).toString('hex'); },
    ...options,
  });
  await bridge.listen();
  onTestFinished(() => bridge.close());
  const socket = bridge.server.address();
  assert.ok(socket && typeof socket !== 'string');
  const url = `http://127.0.0.1:${socket.port}`;
  bridge.setPublicOrigin(url);
  async function request(path: string, data?: unknown, token?: string, headers: Record<string, string> = {}) {
    const response = await fetch(url + path, {
      method: data === undefined ? 'GET' : 'POST',
      headers: { Origin: website, ...(token ? { Authorization: `Bearer ${token}` } : {}),
        ...(data === undefined ? {} : { 'Content-Type': 'application/json' }), ...headers },
      ...(data === undefined ? {} : { body: JSON.stringify(data) }),
    });
    return { status: response.status, headers: response.headers, data: await response.json() };
  }
  async function connect(select = true) {
    const reply = await request('/v1/connect', { code: bridge.pairing.code });
    assert.equal(reply.status, 201);
    if (select) assert.equal((await request('/v1/select', { public_key: address }, reply.data.token)).status, 200);
    return reply.data.token as string;
  }
  return { bridge, url, request, connect, signs: () => signs };
}

test('canonical origins, exact Host, token origin, and preflight preserve authority', async () => {
  for (const origin of ['null', '', 'https://audit.example/', 'https://audit.example:443', 'https://user@audit.example', 'http://audit.example'])
    assert.equal(validOrigin(origin), false, origin);
  for (const origin of ['https://audit.example', 'http://127.0.0.1:1234', 'http://localhost:1234'])
    assert.equal(validOrigin(origin), true, origin);
  const f = await fixture();
  const token = await f.connect();
  for (const origin of ['null', f.url, 'http://evil.example'])
    assert.equal((await f.request('/v1/account', undefined, token, { Origin: origin })).status, 403);
  assert.equal((await f.request('/v1/account', undefined, token, { Origin: 'https://other.example' })).status, 401);
  assert.equal((await f.request('/v1/account', undefined, token, { Host: 'evil.example' })).status, 403);
  assert.equal((await f.request('/v1/signers')).status, 401);
  const preflight = await fetch(f.url + '/v1/requests', { method: 'OPTIONS', headers: { Origin: website } });
  assert.equal(preflight.status, 204);
  assert.equal(preflight.headers.get('access-control-allow-origin'), website);
  assert.equal(preflight.headers.get('access-control-allow-credentials'), null);
  assert.equal(f.signs(), 0);
});

test('two simultaneous uses of one pairing code create only one session', async () => {
  const f = await fixture();
  const code = f.bridge.pairing.code;
  const replies = await Promise.all([f.request('/v1/connect', { code }), f.request('/v1/connect', { code })]);
  assert.deepEqual(replies.map((r) => r.status).sort(), [201, 403]);
});

test('64 live sessions reject a 65th and revocation releases a slot', async () => {
  const f = await fixture();
  const tokens = [];
  for (let i = 0; i < 64; i++) tokens.push(await f.connect(false));
  const code = f.bridge.pairing.code;
  assert.equal((await f.request('/v1/connect', { code })).status, 429);
  await f.request('/v1/disconnect', {}, tokens[0]);
  assert.equal((await f.request('/v1/connect', { code })).status, 201);
});

test('32 active requests reject a 33rd and canceled requests never sign', async () => {
  const f = await fixture({ review: (_record, { signal }) => new Promise((_resolve, reject) => {
    if (signal.aborted) reject(signal.reason);
    else signal.addEventListener('abort', () => reject(signal.reason), { once: true });
  }) });
  const token = await f.connect();
  const payload = input();
  for (let i = 0; i < 32; i++) assert.equal((await f.request('/v1/requests', { ...payload, id: `cap-${i}` }, token)).status, 201);
  assert.equal((await f.request('/v1/requests', { ...payload, id: 'overflow' }, token)).status, 429);
  await f.request('/v1/requests/cap-31/cancel', {}, token);
  assert.equal((await f.request('/v1/requests', { ...payload, id: 'after-cancel' }, token)).status, 201);
  assert.equal(f.signs(), 0);
});

test('1000 completed records reject further records but preserve identical retries', async () => {
  const f = await fixture({ review: async () => false });
  const token = await f.connect();
  const payload = input();
  for (let i = 0; i < 1000; i++) assert.equal((await f.request('/v1/requests', { ...payload, id: `done-${i}` }, token)).status, 201);
  assert.equal((await f.request('/v1/requests', { ...payload, id: 'overflow' }, token)).status, 429);
  assert.equal((await f.request('/v1/requests', { ...payload, id: 'done-0' }, token)).status, 200);
  assert.equal(f.signs(), 0);
}, 15000);

test('1000 early cancellation IDs reject new IDs and block late creates', async () => {
  const f = await fixture();
  const token = await f.connect();
  for (let i = 0; i < 1000; i++) assert.equal((await f.request(`/v1/requests/early-${i}/cancel`, {}, token)).status, 200);
  assert.equal((await f.request('/v1/requests/overflow/cancel', {}, token)).status, 429);
  assert.equal((await f.request('/v1/requests', input('early-0'), token)).status, 409);
  assert.equal(f.signs(), 0);
}, 15000);

test('body limits, JSON shape, and unknown fields reject before signing', async () => {
  const f = await fixture();
  const token = await f.connect();
  assert.equal((await f.request('/v1/requests', { padding: 'x'.repeat(49153) }, token)).status, 413);
  for (const data of [[], null, 1, 'text']) assert.equal((await f.request('/v1/requests', data, token)).status, 400);
  assert.equal((await f.request('/v1/requests', { ...input(), extra: true }, token)).status, 400);
  assert.equal((await f.request('/v1/requests', input(), token, { 'Content-Type': 'text/plain' })).status, 415);
  assert.equal(f.signs(), 0);
});

for (const route of ['/v1/signers', '/v1/select']) test(`revocation during ${route} lookup suppresses authority`, async () => {
  let hold = false, entered = false;
  const gate = Promise.withResolvers<void>();
  const f = await fixture({ listSigners: async () => { if (hold) { entered = true; await gate.promise; } return [{ public_key: address }]; } });
  const token = await f.connect(false);
  hold = true;
  const waiting = f.request(route, route.endsWith('select') ? { public_key: address } : undefined, token);
  await until(() => entered);
  await f.request('/v1/disconnect', {}, token);
  gate.resolve();
  assert.equal((await waiting).status, 401);
  assert.equal(f.signs(), 0);
});

test('XDR restrictions reject extra signatures, fee bumps, preconditions, alternate encoding, and extra operations', () => {
  const good = input();
  assert.equal(inspectTransaction(good, address).details.operation.type, 'manageData');
  const signed = builder().setTimeout(180).build(); signed.sign(key);
  const inner = builder().setTimeout(180).build();
  const cases = [
    ['signed', signed.toXDR()],
    ['fee-bump', TransactionBuilder.buildFeeBumpTransaction(address, '100', inner, Networks.TESTNET).toXDR()],
    ['v2-preconditions', builder().setMinAccountSequence('1').setTimeout(180).build().toXDR()],
    ['extra-operation', builder().addOperation(Operation.manageData({ name: 'second', value: 'offline' })).setTimeout(180).build().toXDR()],
    ['alternate-base64', good.transaction_xdr + '\n'],
    ['trailing-XDR', Buffer.concat([Buffer.from(good.transaction_xdr, 'base64'), Buffer.from([0, 0, 0, 0])]).toString('base64')],
    ['oversize-XDR', 'A'.repeat(32769)],
  ];
  for (const [name, transaction_xdr] of cases) assert.throws(() => inspectTransaction({ ...good, transaction_xdr }, address), undefined, name);
  const signature = Buffer.from(key.sign(inspectTransaction(good, address).tx.hash())).toString('hex');
  assert.ok(attachSignature(good, address, signature));
  assert.throws(() => attachSignature(good, other.publicKey(), signature), /verification/);
  assert.throws(() => attachSignature(good, address, signature.toUpperCase()), /verification/);
  assert.throws(() => inspectTransaction({ ...good, network_passphrase: Networks.PUBLIC }, address), /testnet/);
});

test('time bounds enforce five minutes using the injected clock', () => {
  const now = Math.floor(Date.now() / 1000) * 1000;
  const payload = (min: number, max: number) => ({ ...input(), transaction_xdr: builder().setTimebounds(min, max).build().toXDR() });
  assert.equal(inspectTransaction(payload(0, now / 1000 + 300), address, now).expires, now + 300000);
  for (const [min, max] of [[0, now / 1000], [0, now / 1000 + 301], [now / 1000 + 1, now / 1000 + 100], [0, 0]])
    assert.throws(() => inspectTransaction(payload(min, max), address, now));
});

test('the review hook cannot mutate the recorded account, hash, or operation', async () => {
  const payload = input();
  const expectedHash = inspectTransaction(payload, address).details.hash;
  const f = await fixture({ review: async (record) => {
    record.details.hash = '0'.repeat(64);
    record.signer.public_key = other.publicKey();
    record.details.source = other.publicKey();
    return true;
  } });
  const token = await f.connect();
  await f.request('/v1/requests', payload, token);
  let result: any;
  await until(async () => { result = await f.request('/v1/requests/audit', undefined, token); return result.data.state === 'signed'; });
  assert.equal(result.data.hash, expectedHash);
  const signed = TransactionBuilder.fromXDR(result.data.signed_xdr, Networks.TESTNET);
  assert.equal(signed.source, address);
  assert.ok(key.verify(signed.hash(), signed.signatures[0].signature.toBytes()));
});
