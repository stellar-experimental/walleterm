import { expect, test } from 'bun:test';
import { createBridge } from '/private/tmp/walleterm-audit-40d6cca9db73/bridge/server.ts';

test('distinct forged Origin values reach the shared pairing limit', async () => {
  const bridge = createBridge({ port: 0 });
  await bridge.listen();

  try {
    const address = bridge.server.address();
    if (!address || typeof address === 'string') throw new Error('The bridge did not return a TCP address.');
    const bridgeOrigin = `http://127.0.0.1:${address.port}`;
    bridge.setPublicOrigin(bridgeOrigin);
    const initialCode = bridge.pairing.code;

    for (let index = 0; index < 5; index++) {
      const siteOrigin = `https://forged-${index}.invalid`;
      const response = await fetch(`${bridgeOrigin}/v1/connect`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Origin: siteOrigin },
        body: JSON.stringify({ code: 'wrong' }),
      });

      expect(response.status).toBe(403);
      expect(response.headers.get('access-control-allow-origin')).toBe(siteOrigin);
      await response.arrayBuffer();
    }

    expect(bridge.pairing.code).not.toBe(initialCode);

    const finalOrigin = 'https://forged-final.invalid';
    const locked = await fetch(`${bridgeOrigin}/v1/connect`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Origin: finalOrigin },
      body: JSON.stringify({ code: bridge.pairing.code }),
    });

    expect(locked.status).toBe(429);
    expect(locked.headers.get('access-control-allow-origin')).toBe(finalOrigin);
    await locked.arrayBuffer();
  } finally {
    await bridge.close();
  }
});
