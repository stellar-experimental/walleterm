import assert from 'node:assert/strict';
import { readFileSync, existsSync } from 'node:fs';
import { resolve, dirname, relative } from 'node:path';

const [build, distribution] = process.argv.slice(2);
const parser = new Bun.Transpiler({ loader: 'js' });
const pending = ['sdk/walleterm.js', 'sdk/connect.js', 'sdk/scan.js'].map(name => resolve(distribution, name));
const seen = new Set<string>();
const edges: { from: string; path: string; kind: string; target: string }[] = [];
while (pending.length) {
  const filename = pending.pop()!;
  if (seen.has(filename)) continue;
  assert.ok(existsSync(filename), filename);
  seen.add(filename);
  for (const entry of parser.scanImports(readFileSync(filename, 'utf8'))) {
    assert.ok(entry.path.startsWith('.'), `Unexpected external import: ${entry.path}`);
    const target = resolve(dirname(filename), entry.path);
    assert.ok(target.startsWith(distribution + '/'), target);
    assert.ok(existsSync(target), target);
    edges.push({ from: relative(distribution, filename), ...entry, target: relative(distribution, target) });
    pending.push(target);
  }
}
const dynamic = edges.filter(entry => entry.kind === 'dynamic-import');
assert.equal(dynamic.length, 1);
const decoder = await import(resolve(distribution, dynamic[0].target));
assert.equal(typeof decoder.default, 'function');
const pkg = JSON.parse(readFileSync(resolve(build, 'package.json'), 'utf8'));
const exportsChecked = [];
for (const [name, entry] of Object.entries(pkg.exports)) {
  for (const target of typeof entry === 'string' ? [entry] : Object.values(entry as Record<string, string>)) {
    assert.ok(existsSync(resolve(build, target)), target);
    exportsChecked.push({ name, target });
  }
}
const { createDemoSite } = await import(resolve(build, 'demo/server.ts'));
const site = createDemoSite();
assert.equal(site.server.listening, false);
const assets = [...seen].map(filename => relative(distribution, filename)).concat('sdk/connect.css');
const demo = [];
for (const name of assets) {
  let code = 0, headers: Record<string, string> = {}, bytes: Buffer | undefined;
  site.server.emit('request', { url: '/' + name, method: 'GET', headers: { host: '127.0.0.1:8788' } }, {
    writeHead(value: number, fields: Record<string, string>) { code = value; headers = fields; },
    end(value: Buffer) { bytes = value; },
  });
  assert.equal(code, 200, name);
  assert.equal(headers['Content-Type'], `${name.endsWith('.css') ? 'text/css' : 'text/javascript'}; charset=utf-8`);
  assert.deepEqual(bytes, readFileSync(resolve(distribution, name)), name);
  demo.push({ path: '/' + name, status: code, contentType: headers['Content-Type'] });
}
assert.equal(site.server.listening, false);
console.log(JSON.stringify({ status: 'passed', modules: seen.size, edges, dynamicDecoder: dynamic[0].target,
  decoderExport: typeof decoder.default, exportsChecked, demo, socketsOpened: 0 }, null, 2));
