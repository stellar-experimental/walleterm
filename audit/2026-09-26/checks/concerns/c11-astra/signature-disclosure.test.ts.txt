import { test } from 'bun:test';
import assert from 'node:assert/strict';
import * as sdk from '/private/tmp/walleterm-audit-40d6cca9db73/node_modules/@stellar/stellar-sdk';
import { gAuthorizer, invoke, requireAddressCredentials } from '/private/tmp/walleterm-audit-40d6cca9db73/tests/contracts.ts';
import { simulationError, simulationSuccess } from '/private/tmp/walleterm-audit-40d6cca9db73/tests/simulations.ts';

// Isolated mock keys. This file never imports live-utils or uses a network client.
const pairs = [21, 22, 23].map((n) => sdk.Keypair.fromRawEd25519Seed(Buffer.alloc(32, n)));
const keys = Object.fromEntries(pairs.map((pair, i) => [String.fromCharCode(97 + i), {
  name: String.fromCharCode(97 + i), publicKey: pair.publicKey(), rawPublicKey: pair.rawPublicKey(),
}]));
const target = sdk.StrKey.encodeContract(Buffer.alloc(32, 31));
const callArgs = (amount: number) => [new sdk.Address(keys.b.publicKey).toScVal(), sdk.xdr.ScVal.scvU32(amount)];
const root = (amount: number) => new sdk.xdr.SorobanAuthorizedInvocation({
  function: sdk.xdr.SorobanAuthorizedFunction.sorobanAuthorizedFunctionTypeContractFn(
    new sdk.xdr.InvokeContractArgs({
      contractAddress: new sdk.Address(target).toScAddress(), functionName: 'ping', args: callArgs(amount),
    }),
  ),
  subInvocations: [],
});

test('ordinary invoke discloses a valid changed-tree signature before enforcement failure', async () => {
  const injected = new sdk.xdr.SorobanAuthorizationEntry({
    credentials: sdk.xdr.SorobanCredentials.sorobanCredentialsAddressV2(
      new sdk.xdr.SorobanAddressCredentials({
        address: new sdk.Address(keys.b.publicKey).toScAddress(), nonce: 9n,
        signatureExpirationLedger: 0, signature: sdk.xdr.ScVal.scvVoid(),
      }),
    ),
    rootInvocation: root(999),
  });
  let digestCalls = 0;
  let disclosed: sdk.xdr.SorobanAuthorizationEntry | undefined;
  let enforcementHash: string | undefined;
  const ctx = {
    sdk, keys, networkPassphrase: sdk.Networks.TESTNET,
    async signDigest(_key: unknown, payload: Uint8Array) {
      digestCalls++;
      return pairs[1].sign(payload);
    },
    async sign() { assert.fail('No envelope signature is permitted.'); },
    async send() { assert.fail('No transaction submission is permitted.'); },
    rpc: {
      async getAccount() { return new sdk.Account(keys.a.publicKey, '1'); },
      async simulateTransaction(tx: sdk.Transaction, _resources: unknown, mode: string) {
        if (mode === 'record') return simulationSuccess({
          latestLedger: 100, result: { auth: [injected], retval: sdk.xdr.ScVal.scvU32(1) },
        });
        assert.equal(mode, 'enforce');
        // Decode the same serialized envelope that the installed RPC client sends.
        const decoded = sdk.TransactionBuilder.fromXDR(tx.toXDR(), sdk.Networks.TESTNET);
        assert.ok(decoded instanceof sdk.Transaction);
        const op = decoded.operations[0];
        assert.equal(op.type, 'invokeHostFunction');
        disclosed = op.auth?.[0];
        enforcementHash = Buffer.from(decoded.hash()).toString('hex');
        return simulationError('HostError: Error(Auth, InvalidAction)', { latestLedger: 100 });
      },
    },
  };
  // No negative-test option or mutation hook enables this path.
  await assert.rejects(invoke(ctx, {
    contractId: target, method: 'ping', args: callArgs(1), label: 'ordinary-call',
    authorizers: [gAuthorizer(ctx, keys.b)],
  }), /enforce simulation failed/);
  assert.equal(digestCalls, 1);
  assert.ok(disclosed);
  const credentials = requireAddressCredentials(disclosed);
  assert.equal(credentials.signatureExpirationLedger, 160);
  assert.equal(disclosed.rootInvocation.toXdr('base64'), root(999).toXdr('base64'));
  const signature = sdk.scValToNative(credentials.signature)[0].signature;
  const digestFor = (entry: sdk.xdr.SorobanAuthorizationEntry, network = sdk.Networks.TESTNET) =>
    sdk.hash(sdk.buildAuthorizationEntryPreimage(entry, 160, network).toXdr());
  const publicKey = sdk.Keypair.fromPublicKey(keys.b.publicKey);
  assert.ok(publicKey.verify(digestFor(disclosed), signature));
  assert.equal(publicKey.verify(digestFor(disclosed, sdk.Networks.PUBLIC), signature), false);
  const intended = sdk.xdr.SorobanAuthorizationEntry.fromXdr(disclosed.toXdr());
  intended.rootInvocation = root(1);
  assert.equal(publicKey.verify(digestFor(intended), signature), false);
  const otherEnvelope = new sdk.TransactionBuilder(new sdk.Account(keys.c.publicKey, '77'), {
    fee: '100', networkPassphrase: sdk.Networks.TESTNET,
  }).addOperation(new sdk.Contract(target).call('ping', ...callArgs(999))).setTimeout(30).build();
  assert.notEqual(Buffer.from(otherEnvelope.hash()).toString('hex'), enforcementHash);
  const op = otherEnvelope.operations[0];
  assert.equal(op.type, 'invokeHostFunction');
  const rebuilt = new sdk.TransactionBuilder(new sdk.Account(keys.c.publicKey, '77'), {
    fee: '100', networkPassphrase: sdk.Networks.TESTNET,
  }).addOperation(sdk.Operation.invokeHostFunction({ func: op.func, auth: [disclosed] })).setTimeout(30).build();
  const attached = rebuilt.operations[0];
  assert.equal(attached.type, 'invokeHostFunction');
  assert.ok(publicKey.verify(digestFor(attached.auth![0]), signature));
  console.log(JSON.stringify({
    check: 'c11-disclosure', digest_signatures: digestCalls, envelope_signatures: 0, submissions: 0,
    original_call_rejected: true, changed_tree_signature_valid_after_rejection: true,
    different_envelope_signature_valid: true, mainnet_signature_valid: false,
    intended_tree_signature_valid: false, expiration: credentials.signatureExpirationLedger,
    protocol_acceptance: 'not_run',
  }));
});
