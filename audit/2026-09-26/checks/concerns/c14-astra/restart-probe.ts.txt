// Offline audit only. The pipeline body uses injected commands, RPC, and a mock signer.
import assert from 'node:assert/strict';
import { execFileSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import { readFileSync, mkdtempSync, rmSync, existsSync } from 'node:fs';
import { join } from 'node:path';
import { createSubmissionGuard } from '/private/tmp/walleterm-audit-40d6cca9db73/tests/submission.ts';

const root = '/private/tmp/walleterm-audit-40d6cca9db73';
const revision = '40d6cca9db732a0db16d154c80d4a153bf33c6b7';
const makeGuard = (directory: string) => {
  let lookups = 0;
  const guard = createSubmissionGuard({
    directory, networkPassphrase: 'offline', record() {},
    rpc: {
      async sendTransaction() { throw Error('Unexpected shared send'); },
      async getTransaction() { lookups++; throw Error('Unexpected lookup'); },
    },
  });
  return { guard, lookupCount: () => lookups };
};

if (process.argv[2] === 'child') {
  const { guard, lookupCount } = makeGuard(process.argv[3]!);
  // This matches the pre-sign gate at live-utils.ts:53. No signer process runs.
  guard.assertClear();
  const mockSignerCalls = 1;
  assert.equal(await guard.reconcile(), null);
  console.log(JSON.stringify({ pid: process.pid, mockSignerCalls, reconcile: null, lookups: lookupCount() }));
} else {
  const hashes = Object.fromEntries(['tests/cli-pipeline.ts', 'tests/live-utils.ts', 'tests/submission.ts'].map(file => {
    const bytes = readFileSync(join(root, file));
    const baseline = execFileSync('git', ['show', `${revision}:${file}`], {
      cwd: '/Users/kalepail/Desktop/walleterm-v2',
    });
    assert.deepEqual(bytes, baseline);
    return [file, createHash('sha256').update(bytes).digest('hex')];
  }));
  const file = join(root, 'tests/cli-pipeline.ts');
  const body = new Bun.Transpiler({ loader: 'ts' }).transformSync(readFileSync(file, 'utf8'))
    .replace(/^import .*\n/gm, '')
    .replaceAll('import.meta.url', JSON.stringify(new URL('file://' + file).href));
  assert.equal(/\bimport\b/.test(body), false);
  const AsyncFunction = Object.getPrototypeOf(async function () {}).constructor;
  const run = new AsyncFunction('assert', 'execFileSync', 'writeFileSync', 'ctx', body);
  const scratch = mkdtempSync(join(import.meta.dir, 'state-'));
  const cases = [];
  try {
    for (const fault of ['send_timeout', 'lookup_not_found', 'lookup_transport_error']) {
      const directory = join(scratch, fault);
      const { guard } = makeGuard(directory);
      const hash = 'ab'.repeat(32);
      let signerCalls = 0, lookups = 0, sends = 0;
      const ctx = {
        keys: { a: { publicKey: 'GMOCKA', rawPublicKey: Buffer.alloc(32) }, c: { publicKey: 'GMOCKC' } },
        networkPassphrase: 'offline',
        sdk: { TransactionBuilder: { fromXDR: () => ({ hash: () => Buffer.from(hash, 'hex') }) } },
        async signDigest() { guard.assertClear(); signerCalls++; return Buffer.alloc(64); },
        rpc: { async getTransaction(requested: string) {
          assert.equal(requested, hash); lookups++;
          if (fault === 'lookup_transport_error') throw Error('Mock lookup transport failure');
          return { status: 'NOT_FOUND' };
        } },
        record() { throw Error('Unexpected passing record'); },
      };
      const command = (name: string, args: string[], options: { input?: string; timeout: number }) => {
        assert.equal(name, 'stellar');
        assert.equal(options.timeout, 60000);
        if (args[1] === 'new') return 'mock-unsigned';
        if (args[1] === 'hash') return hash;
        if (args[1] === 'decode') return JSON.stringify({ tx: { signatures: [] } });
        if (args[1] === 'encode') return 'mock-signed';
        assert.deepEqual(args, ['tx', 'send', '--network', 'testnet']);
        assert.equal(options.input, 'mock-signed');
        assert.equal(guard.pending(), null);
        sends++;
        if (fault === 'send_timeout') throw Error('Mock timeout after possible acceptance');
        // Deliberately opaque: no assumption about actual CLI stdout or polling semantics.
        return 'mock CLI returned successfully';
      };
      let observedError = '';
      try {
        await run(assert, command, () => {}, ctx);
        assert.fail('Expected the injected failure');
      } catch (error) { observedError = String(error); }
      assert.equal(sends, 1);
      assert.equal(signerCalls, 1);
      assert.equal(lookups, fault === 'send_timeout' ? 0 : 1);
      assert.equal(existsSync(join(directory, 'pending-submission.json')), false);
      const next = JSON.parse(execFileSync(process.execPath, [import.meta.filename, 'child', directory], {
        cwd: import.meta.dir, encoding: 'utf8', timeout: 10000,
      }));
      assert.notEqual(next.pid, process.pid);
      assert.equal(next.mockSignerCalls, 1);
      assert.equal(next.lookups, 0);
      cases.push({ fault, observedError, pendingAtSend: false, pendingAfterFailure: false,
        sends, signerCalls, originalHashLookups: lookups, nextProcess: next });
    }
    console.log(JSON.stringify({ revision, hashes, bun: Bun.version, platform: process.platform,
      liveCommands: 0, networkRequests: 0, realSignatures: 0, cases }, null, 2));
  } finally { rmSync(scratch, { recursive: true, force: true }); }
}
