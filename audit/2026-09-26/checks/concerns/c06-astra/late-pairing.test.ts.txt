import { onTestFinished, test } from 'bun:test';
import assert from 'node:assert/strict';
import { WalletermConnect } from '/private/tmp/walleterm-audit-40d6cca9db73/sdk/connect.ts';
import { WalletermClient } from '/private/tmp/walleterm-audit-40d6cca9db73/sdk/walleterm.ts';

// Preserve the real component and client control flow. Mock only browser IO and
// the user's completed wallet choice. Every HTTP request stays in memory.
for (const pause of ['selection-response', 'previous-disconnect-response']) {
  test(`destroy permits a late commit after ${pause}`, async () => {
    const reached = Promise.withResolvers<void>();
    const release = Promise.withResolvers<void>();
    const oldToken = 'o'.repeat(43);
    const nextToken = 'n'.repeat(43);
    const key = 'c06-mock-session';
    const address = 'GMOCK_SELECTED_WALLET';
    const calls: { route: string; afterDestroy: boolean; previous: boolean }[] = [];
    const changes: { afterDestroy: boolean; client: WalletermClient | null }[] = [];
    const writes: { afterDestroy: boolean; value: string }[] = [];
    let ui: WalletermConnect;
    let heldSignal: AbortSignal | null | undefined;
    const fetcher = async (input: RequestInfo | URL, options?: RequestInit) => {
      const url = new URL(String(input));
      assert.equal(url.origin, 'https://c06.invalid');
      const route = url.pathname;
      const previous = new Headers(options?.headers).get('Authorization') === `Bearer ${oldToken}`;
      calls.push({ route, afterDestroy: ui.destroyed, previous });
      if (
        (pause === 'selection-response' && route === '/v1/select') ||
        (pause === 'previous-disconnect-response' && route === '/v1/disconnect' && previous)
      ) {
        heldSignal = options?.signal;
        reached.resolve();
        await new Promise<void>((resolve, reject) => {
          const stop = () => reject(heldSignal?.reason);
          if (heldSignal?.aborted) return stop();
          heldSignal?.addEventListener('abort', stop, { once: true });
          release.promise.then(() => {
            heldSignal?.removeEventListener('abort', stop);
            resolve();
          });
        });
      }
      switch (route) {
        case '/v1/connect':
          return Response.json({ token: nextToken, wallet_scope: 'available', selection_revision: 0 });
        case '/v1/signers':
          return Response.json({ signers: [{ public_key: address }], grant_id: 'mock-grant' });
        case '/v1/select':
          return Response.json({ public_key: address, network_passphrase: 'testnet', selection_revision: 1 });
        case '/v1/disconnect':
          return Response.json({ disconnected: true });
        default:
          throw Error(`Unexpected mock request: ${route}`);
      }
    };
    const globals = {
      fetch: fetcher,
      document: { removeEventListener() {} },
      removeEventListener() {},
      sessionStorage: {
        setItem(name: string, value: string) {
          assert.equal(name, key);
          writes.push({ afterDestroy: ui.destroyed, value });
        },
        removeItem() { throw Error('Unexpected storage removal'); },
      },
    };
    for (const [name, value] of Object.entries(globals)) {
      const original = Object.getOwnPropertyDescriptor(globalThis, name);
      Object.defineProperty(globalThis, name, { value, configurable: true });
      onTestFinished(() => original
        ? Object.defineProperty(globalThis, name, original)
        : Reflect.deleteProperty(globalThis, name));
    }
    const nodes = new Map();
    const node = (name: string) => {
      if (!nodes.has(name)) nodes.set(name, {
        value: name === 'url' ? 'https://c06.invalid' : name === 'code' ? '00123456' : '',
        hidden: false, classList: { remove() {} }, setAttribute() {}, focus() {},
      });
      return nodes.get(name);
    };
    const previous = new WalletermClient('https://c06.invalid');
    previous.token = oldToken;
    previous.account = { address: 'GMOCK_PREVIOUS_WALLET', networkPassphrase: 'testnet' };
    ui = Object.assign(Object.create(WalletermConnect.prototype), {
      client: previous, account: previous.account, destroyed: false,
      scanning: null, connection: null, working: false, busy: false,
      state: 'connected', sessionStorageKey: key, $: node,
      update() {}, message() {},
      chooseWallet: async () => address,
      onChange(value: { client: WalletermClient | null }) {
        changes.push({ afterDestroy: ui.destroyed, client: value.client });
      },
      dialog: { open: true, close() { this.open = false; } },
      trigger: node('trigger'),
    });
    const pairing = ui.connect();
    try {
      await reached.promise;
      const controller = ui.connection;
      assert.ok(controller);
      ui.destroy();
      assert.equal(controller.signal.aborted, false);
      assert.equal(heldSignal?.aborted, false);
      assert.equal(previous.token, oldToken);
      assert.equal(changes.length, 0);
      assert.equal(writes.length, 0);
      release.resolve();
      await pairing;
      assert.equal(ui.destroyed, true);
      assert.equal(previous.token, null);
      assert.notEqual(ui.client, previous);
      assert.equal(ui.client?.token, nextToken);
      assert.equal(ui.account?.address, address);
      assert.equal(changes.length, 1);
      assert.equal(changes[0]?.afterDestroy, true);
      assert.equal(changes[0]?.client, ui.client);
      assert.equal(writes.length, 1);
      assert.equal(writes[0]?.afterDestroy, true);
      assert.equal(JSON.parse(writes[0]!.value).token, nextToken);
      const revocation = calls.find(call => call.route === '/v1/disconnect' && call.previous);
      assert.equal(revocation?.afterDestroy, pause === 'selection-response');
      assert.equal(calls.some(call => call.route.startsWith('/v1/requests')), false);
      console.log(JSON.stringify({
        pause, destroyed: ui.destroyed, aborted: controller.signal.aborted,
        lateSavedSession: true, lateConnectionCallback: true,
        previousDisconnectStartedAfterDestroy: revocation?.afterDestroy,
        signingRequests: 0, calls,
      }));
    } finally {
      release.resolve();
      await pairing;
    }
  });
}
