import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync } from 'node:fs';

const sourcePath = '/private/tmp/walleterm-audit-40d6cca9db73/tests/cap85.ts';
const source = readFileSync(sourcePath, 'utf8');
const reconciliation = source.slice(
  source.indexOf('export async function reconcileInflight('),
  source.indexOf('export async function prepareCap85('),
);
const stepper = source.slice(source.indexOf('const stepper ='), source.indexOf('// The saved hash of an uploaded artifact.'));
const x06 = source.slice(source.indexOf('async function x06('), source.indexOf('// Observation only:'));

assert.ok(reconciliation.startsWith('export async function reconcileInflight('));
assert.ok(stepper.startsWith('const stepper ='));
assert.ok(x06.startsWith('async function x06('));

const executable = new Bun.Transpiler({ loader: 'ts' }).transformSync(
  `${reconciliation}\n${stepper}\n${x06}`.replaceAll('export ', ''),
);

type Mode = 'control' | 'saved-success' | 'inflight-success';

async function check(mode: Mode) {
  let chain = mode === 'control' ? { type: 'external_ref', version: 2 } : { type: 'wasm', version: 1 };
  let writes = 0;
  let operations = 0;
  let transactionQueries = 0;
  const records: Array<{ id: string; outcome: string }> = [];
  const state: {
    contracts: Record<string, { id: string }>;
    wasm: Record<string, { hash: string }>;
    done: Record<string, object>;
    steps: Record<string, object>;
    inflight?: Record<string, string>;
  } = {
    contracts: { manager: { id: 'mock-manager' } },
    wasm: { 'target-v1.wasm': { hash: 'ab'.repeat(32) } },
    done: {},
    steps: { 'X06:adopt-ref': { hash: 'mock-ref-hash' } },
  };
  if (mode === 'saved-success') state.steps['X06:adopt-wasm'] = { hash: 'mock-wasm-hash' };
  if (mode === 'inflight-success') {
    state.inflight = {
      row: 'X06',
      label: 'adopt-wasm',
      key: 'X06:adopt-wasm',
      hash: 'mock-wasm-hash',
    };
  }

  const ctx = {
    sdk: {},
    keys: { b: { publicKey: 'mock-admin' } },
    rpc: {
      getTransaction: async () => {
        transactionQueries += 1;
        return { status: 'SUCCESS', ledger: 42 };
      },
    },
    record: (id: string, outcome: string) => records.push({ id, outcome }),
  };
  const dependencies = {
    saveState: () => {
      writes += 1;
    },
    readU32: async () => chain.version,
    UnknownSubmission: class extends Error {},
    deployWasm: async () => 'mock-plain',
    addr: (_sdk: unknown, value: string) => value,
    str: (_sdk: unknown, value: string) => value,
    bytes: (_sdk: unknown, value: Uint8Array) => value,
    readInstanceExecutable: async () => ({ type: chain.type }),
    readOptionalHash: async () => 'mock-resolved-hash',
    gAuthorizer: () => 'mock-authorizer',
    call: (_sdk: unknown, _contract: string, method: string) => method,
    wasmHash: () => 'ab'.repeat(32),
    invokeOperation: async (_ctx: unknown, _state: unknown, options: { operation: string }) => {
      assert.equal(options.operation, 'adopt_wasm');
      operations += 1;
      chain = { type: 'wasm', version: 1 };
      return { hash: 'mock-wasm-hash' };
    },
    FILES: { v1: 'target-v1.wasm' },
    TAG: 'target',
  };
  const run = new Function(
    ...Object.keys(dependencies),
    `${executable}; return { reconcileInflight, x06 };`,
  )(...Object.values(dependencies));

  if (mode === 'inflight-success') await run.reconcileInflight(ctx, state);
  let error: string | undefined;
  let returned = false;
  try {
    await run.x06(ctx, state, {});
    returned = true;
  } catch (value) {
    error = String(value);
  }

  if (mode === 'control') {
    assert.equal(error, undefined);
    assert.equal(returned, true);
    assert.equal(operations, 1);
    assert.equal(transactionQueries, 0);
  } else {
    assert.match(error ?? '', /adopt_ref did not move the contract onto the v2 reference/);
    assert.equal(returned, false);
    assert.equal(operations, 0);
    assert.ok(state.steps['X06:adopt-wasm']);
    assert.equal(state.inflight, undefined);
    assert.equal(Object.hasOwn(state.done, 'X06'), false);
  }
  if (mode === 'saved-success') {
    assert.equal(writes, 0);
    assert.equal(transactionQueries, 0);
  }
  if (mode === 'inflight-success') {
    assert.equal(writes, 1);
    assert.equal(transactionQueries, 1);
  }

  return {
    mode,
    error,
    operations,
    transaction_queries: transactionQueries,
    checkpoint_writes: writes,
    final_step_cached: Boolean(state.steps['X06:adopt-wasm']),
    inflight_retained: Boolean(state.inflight),
    x06_returned: returned,
    records,
  };
}

console.log(
  JSON.stringify(
    {
      status: 'passed',
      concern_confirmed: true,
      frozen_source_sha256: createHash('sha256').update(source).digest('hex'),
      control: await check('control'),
      saved_success: await check('saved-success'),
      reconciled_success: await check('inflight-success'),
      limits: 'Exact frozen reconciliation, stepper, and X06 bodies. All external effects use isolated mocks.',
    },
    null,
    2,
  ),
);
