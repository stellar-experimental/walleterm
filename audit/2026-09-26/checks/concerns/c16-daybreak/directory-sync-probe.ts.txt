import {
  closeSync,
  fsyncSync,
  mkdtempSync,
  openSync,
  rmSync,
  unlinkSync,
  writeFileSync,
} from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

const root = mkdtempSync(join(tmpdir(), 'c16-daybreak-bun-'));
const gate = join(root, 'pending-submission.json');
const results: Record<string, string> = {};

try {
  const gateFd = openSync(gate, 'wx', 0o600);
  try {
    writeFileSync(gateFd, '{"pending":true}\n');
    fsyncSync(gateFd);
    results.file_create_fsync = 'passed';
  } finally {
    closeSync(gateFd);
  }

  const createDirFd = openSync(root, 'r');
  try {
    fsyncSync(createDirFd);
    results.directory_fsync_after_create = 'passed';
  } finally {
    closeSync(createDirFd);
  }

  unlinkSync(gate);
  const deleteDirFd = openSync(root, 'r');
  try {
    fsyncSync(deleteDirFd);
    results.directory_fsync_after_unlink = 'passed';
  } finally {
    closeSync(deleteDirFd);
  }

  console.log(JSON.stringify({ bun: Bun.version, platform: process.platform, temp_parent: tmpdir(), results }));
} finally {
  rmSync(root, { recursive: true });
}
