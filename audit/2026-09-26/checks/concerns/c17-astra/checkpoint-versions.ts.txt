// C17 only. Frozen imports, synthetic checkpoints, and no network or signer access.
import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { mkdtempSync, readFileSync, rmSync } from 'node:fs';
import { join } from 'node:path';

const root = '/private/tmp/walleterm-audit-40d6cca9db73';
const audit = '/Users/kalepail/Desktop/walleterm-v2/audit/2026-09-26';
const hash = (bytes) => createHash('sha256').update(bytes).digest('hex');
const manifest = JSON.parse(readFileSync(join(audit, 'manifest.json'), 'utf8'));
for (const [file, expected] of Object.entries(manifest.file_sha256)) {
  assert.equal(hash(readFileSync(join(root, file))), expected, file);
}
const sdk = await import(join(root, 'node_modules/@stellar/stellar-sdk/lib/esm/index.js'));
const cap71 = await import(join(root, 'tests/cap71.ts'));
const cap85 = await import(join(root, 'tests/cap85.ts'));
const directory = mkdtempSync('/private/tmp/c17-astra-');
const keys = Object.fromEntries(['a', 'b', 'c'].map((name, i) => {
  const rawPublicKey = Buffer.alloc(32, i + 1);
  return [name, { name, rawPublicKey, publicKey: sdk.StrKey.encodeEd25519PublicKey(rawPublicKey) }];
}));
const denied = () => assert.fail('No RPC, signature, or submission is allowed here');
const events = [];
// Use the frozen recorder body with memory-only file and console sinks.
const liveSource = readFileSync(join(root, 'tests/live-utils.ts'), 'utf8');
const recordSource = liveSource.slice(liveSource.indexOf('function record('), liveSource.indexOf('\nasync function signDigest'));
const transpile = (source) => new Bun.Transpiler({ loader: 'ts' }).transformSync(source);
const record = new Function('results', 'writeFileSync', 'resultFile', 'console',
  `${transpile(recordSource)}; return record;`)(events, () => {}, 'memory-only', { log() {} });
const common = {
  sdk, keys, networkPassphrase: sdk.Networks.TESTNET, record,
  assertClear() {}, sign: denied, signDigest: denied, send: denied,
};
const results = [];

try {
  const wasmNames = ['cap71_delegate.wasm', 'cap71_raw_account.wasm', 'cap71_target.wasm'];
  const binding = {
    network: common.networkPassphrase, keys: Object.values(keys).map((key) => key.publicKey),
    artifacts: Object.fromEntries(wasmNames.map((file) => [file, hash(readFileSync(join(root,
      'fixtures/cap71/target/wasm32v1-none/release', file)))])),
  };
  const checkpoint = join(directory, 'cap71.json');
  const state = cap71.loadCheckpoint(checkpoint, binding);
  for (const file of wasmNames) state.steps[`CAP71-upload-${file}`] = { protocol: 28 };
  const contracts = {};
  for (const [i, name] of ['target', 'any', 'both', 'weighted', 'chain', 'raw1', 'raw2'].entries()) {
    const address = sdk.StrKey.encodeContract(Buffer.alloc(32, 10 + i));
    contracts[name] = address;
    state.salts[name] = Buffer.alloc(32, 30 + i).toString('hex');
    state.steps[`CAP71-deploy-${name}`] = {
      protocol: 28, retval_xdr: new sdk.Address(address).toScVal().toXdr('base64'),
    };
  }
  state.checks['CAP71-direct-a'] = { protocol: 28, outcome: 'submitted', state_before: 0, state_after: 1 };
  state.checks['CAP71-direct-c'] = { protocol: 28, outcome: 'submitted', state_before: 1, state_after: 2 };
  state.done['CAP71-01'] = { protocol: 28, contracts, checks: {
    a: state.checks['CAP71-direct-a'], c: state.checks['CAP71-direct-c'],
  } };
  cap71.saveCheckpoint(checkpoint, state);
  let protocol = 29; // A hypothetical protocol change, not a claim about testnet.
  let simulations = 0;
  const context71 = { ...common, rows: ['CAP71-01'], cap71Checkpoint: checkpoint,
    rpc: { getVersionInfo: async () => ({ protocolVersion: protocol }), getAccount: denied,
      simulateTransaction() { simulations++; return denied(); } },
  };
  await cap71.runCap71(context71);
  const completed = events.findLast((event) => event.id === 'CAP71-01');
  assert.equal(completed.status, 'passed_previous_run');
  assert.equal(completed.protocol, 28);
  assert.equal(completed.reused_evidence, true);
  assert.equal(simulations, 0);
  results.push({ check: 'cap71_completed_row_after_protocol_change', outcome: 'passed', event: completed });

  // Natural interruption boundary: call() saved both checks, but runCap71() has not saved done yet.
  delete state.done['CAP71-01'];
  cap71.saveCheckpoint(checkpoint, state);
  events.length = 0;
  await cap71.runCap71(context71);
  const resumed = events.findLast((event) => event.id === 'CAP71-01');
  assert.equal(resumed.status, 'passed');
  assert.equal(resumed.protocol, 29);
  assert.equal(resumed.checks.a.protocol, 28);
  assert.equal(resumed.checks.c.protocol, 28);
  assert.equal(resumed.reused_evidence, undefined);
  assert.equal(simulations, 0);
  results.push({ check: 'cap71_incomplete_row_after_protocol_change', outcome: 'passed_reproduction', event: resumed });

  protocol = 26;
  await assert.rejects(cap71.runCap71(context71), /protocol 27 or later/);
  assert.throws(() => cap71.loadCheckpoint(checkpoint, { ...binding, network: 'other' }), /changed/);
  results.push({ check: 'cap71_minimum_protocol_and_binding_controls', outcome: 'passed' });

  const source85 = readFileSync(join(root, 'tests/cap85.ts'), 'utf8');
  const runnerSource = source85.slice(source85.indexOf('export async function runCap85('),
    source85.indexOf('// ---------- offline self-test ----------')).replace('export async', 'async');
  const base = { contracts: Object.fromEntries(['oz_basic_a', 'simple_account_b', 'ed25519_verifier']
    .map((name) => [name, { id: sdk.StrKey.encodeContract(Buffer.alloc(32, 80)) }])) };
  const manifest85 = cap85.loadManifest();
  const file85 = join(directory, 'cap85.json');
  const state85 = cap85.loadState(common, base, manifest85, file85);
  state85.done.X01 = { title: 'synthetic prior X01 evidence' };
  state85.done.X07 = { title: 'synthetic prior observation' };
  cap85.saveState(state85);
  let assertionCalls = 0;
  const rows = [
    { id: 'X01', async run() { assertionCalls++; throw new Error('current assertion reached'); } },
    { id: 'X07', status: 'observed', run: denied },
  ];
  const dependencies = {
    selectRows: cap85.selectRows, loadManifest: cap85.loadManifest, loadBase: () => base,
    loadState: (ctx, b, m) => cap85.loadState(ctx, b, m, file85), prepareCap85: cap85.prepareCap85,
    ROWS: rows, saveState: cap85.saveState, isUnknown: () => false, requestError: (error) => error,
  };
  const run85 = new Function(...Object.keys(dependencies), `${transpile(runnerSource)}; return runCap85;`)
    (...Object.values(dependencies));
  const context85 = { ...common, rows: ['X01', 'X07'], fund: async () => {}, reconcile: async () => null,
    rpc: { getNetwork: async () => ({ passphrase: sdk.Networks.TESTNET, protocolVersion: 29 }) } };
  events.length = 0;
  await run85(context85);
  assert.equal(assertionCalls, 0);
  assert.equal(events.find((event) => event.id === 'X01').status, 'passed_previous_run');
  assert.equal(events.find((event) => event.id === 'X01').reused_evidence, true);
  assert.equal(events.find((event) => event.id === 'X07').status, 'observed_previous_run');
  results.push({ check: 'cap85_completed_rows_skip_current_assertion_and_preserve_historical_status', outcome: 'passed', events: [...events] });
  delete state85.done.X01;
  cap85.saveState(state85);
  await assert.rejects(run85(context85), /current assertion reached/);
  assert.equal(assertionCalls, 1);
  results.push({ check: 'cap85_no_completed_row_reaches_current_assertion', outcome: 'passed' });
  assert.throws(() => cap85.loadState({ ...common, networkPassphrase: 'other' }, base, manifest85, file85), /binding mismatch/);
  await assert.rejects(run85({ ...context85, rpc: { getNetwork: async () => ({ passphrase: sdk.Networks.TESTNET, protocolVersion: 27 }) } }), /protocol 28 or later/);
  results.push({ check: 'cap85_minimum_protocol_and_binding_controls', outcome: 'passed' });

  const stepperSource = source85.slice(source85.indexOf('const stepper ='), source85.indexOf('// The saved hash of an uploaded artifact.'));
  const stepper = new Function('reconcileInflight', 'saveState', `${transpile(stepperSource)}; return stepper;`)
    (cap85.reconcileInflight, cap85.saveState);
  state85.steps['X01:stale-version'] = { status: 'simulation_rejected' };
  const details = { title: 'synthetic incomplete row' };
  events.length = 0;
  await stepper(context85, state85, 'X01', details)('stale-version', denied);
  assert.deepEqual(details.reused_steps, ['stale-version']);
  assert.equal(events[0].status, 'passed_previous_run');
  results.push({ check: 'cap85_incomplete_row_marks_reused_assertion_step', outcome: 'passed', details, events: [...events] });

  console.log(JSON.stringify({ status: 'passed', revision: manifest.revision, bun: Bun.version,
    verified_tracked_files: Object.keys(manifest.file_sha256).length, checks: results,
    scope: 'Actual CAP71 runner; extracted frozen CAP85 runner and stepper; actual recorder with memory-only sinks.',
    limits: 'Synthetic historical records and hypothetical protocol 29; no live state, protocol execution, signatures, or external requests.',
  }, null, 2));
} finally {
  rmSync(directory, { recursive: true, force: true });
}
