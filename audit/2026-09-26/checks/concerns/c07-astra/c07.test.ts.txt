import { onTestFinished, test } from 'bun:test';
import assert from 'node:assert/strict';
import { WalletermClient } from '/private/tmp/walleterm-audit-40d6cca9db73/sdk/walleterm.ts';
import { WalletermConnect } from '/private/tmp/walleterm-audit-40d6cca9db73/sdk/connect.ts';

const token = 'S'.repeat(43);
const account = { address: 'GMOCK', networkPassphrase: 'testnet' };
const key = 'c07-mock-session';

function replaceGlobal(name: string, value: unknown) {
  const descriptor = Object.getOwnPropertyDescriptor(globalThis, name);
  Object.defineProperty(globalThis, name, { configurable: true, writable: true, value });
  onTestFinished(() => {
    if (descriptor) Object.defineProperty(globalThis, name, descriptor);
    else Reflect.deleteProperty(globalThis, name);
  });
}

function fixture(fetcher: typeof fetch, restoring = false) {
  const storage = new Map<string, string>();
  storage.set(key, JSON.stringify({ version: 1, url: 'https://bridge.example', token }));
  replaceGlobal('sessionStorage', {
    getItem: (name: string) => storage.get(name) ?? null,
    setItem: (name: string, value: string) => storage.set(name, value),
    removeItem: (name: string) => storage.delete(name),
  });
  replaceGlobal('fetch', fetcher);
  const client = new WalletermClient('https://bridge.example', { fetch: fetcher, page: null });
  client.token = token;
  client.account = { ...account };
  const changes: unknown[] = [];
  const states: string[] = [];
  // Keep real state, storage, restoration, and health methods. Stub rendering only.
  const ui: WalletermConnect = Object.assign(Object.create(WalletermConnect.prototype), {
    sessionStorageKey: key,
    client: restoring ? null : client,
    account: restoring ? null : { ...account },
    state: restoring ? 'disconnected' : 'connected',
    wallets: [],
    destroyed: false,
    working: false,
    refreshing: false,
    checking: false,
    onChange: (value: unknown) => changes.push(value),
    onStateChange: (value: string) => states.push(value),
    onBusyChange() {},
    update() {},
    hideMenu() {},
    $: () => ({ value: '' }),
  });
  return { ui, client, storage, changes, states };
}

for (const restoring of [false, true]) {
  test(`HTML 401 retains credentials and reports expired; restoring=${restoring}`, async () => {
    const requests: string[] = [];
    const f = fixture(async (url) => {
      requests.push(String(url));
      return new Response('<html>expired</html>', { status: 401 });
    }, restoring);
    if (restoring) await f.ui.restoreSession();
    else await f.ui.checkHealth();
    assert.equal(f.ui.state, 'expired');
    assert.equal(f.ui.client?.token, token);
    assert.equal(f.storage.has(key), true);
    assert.deepEqual(f.ui.account, restoring ? null : account);
    assert.deepEqual(f.changes, []);
    assert.deepEqual(requests, ['https://bridge.example/v1/account']);
  });

  test(`JSON 401 clears credentials and publishes disconnection; restoring=${restoring}`, async () => {
    const f = fixture(async () => Response.json({ error: { message: 'Expired' } }, { status: 401 }), restoring);
    if (restoring) await f.ui.restoreSession();
    else await f.ui.checkHealth();
    assert.equal(f.ui.state, 'expired');
    assert.equal(f.ui.client, null);
    assert.equal(f.ui.account, null);
    assert.equal(f.storage.has(key), false);
    assert.deepEqual(f.changes, [{ client: null, account: null }]);
  });
}

test('late HTML and JSON 401 responses preserve a newer token and account', async () => {
  for (const json of [false, true]) {
    const pending = Promise.withResolvers<Response>();
    const client = new WalletermClient('https://bridge.example', { fetch: () => pending.promise, page: null });
    client.token = token;
    const request = client.getAddress();
    client.token = 'N'.repeat(43);
    client.account = { ...account };
    pending.resolve(json
      ? Response.json({ error: { message: 'Expired' } }, { status: 401 })
      : new Response('<html>expired</html>', { status: 401 }));
    await assert.rejects(request, (error: unknown) => error instanceof Error && Reflect.get(error, 'status') === 401);
    assert.equal(client.token, 'N'.repeat(43));
    assert.deepEqual(client.account, account);
  }
});

test('a fetch rejection preserves credentials and reports unreachable', async () => {
  const f = fixture(async () => { throw new TypeError('Failed to fetch'); });
  await f.ui.checkHealth();
  assert.equal(f.ui.state, 'unreachable');
  assert.equal(f.ui.client?.token, token);
  assert.deepEqual(f.ui.account, account);
  assert.equal(f.storage.has(key), true);
  assert.deepEqual(f.changes, []);
});

test('explicit SDK disconnect and UI sync clear an HTML 401 session', async () => {
  const f = fixture(async () => new Response('<html>expired</html>', { status: 401 }));
  await f.client.disconnect();
  f.ui.sync();
  assert.equal(f.client.token, null);
  assert.equal(f.client.account, null);
  assert.equal(f.ui.client, null);
  assert.equal(f.ui.account, null);
  assert.equal(f.storage.has(key), false);
  assert.deepEqual(f.changes, [{ client: null, account: null }]);
});
