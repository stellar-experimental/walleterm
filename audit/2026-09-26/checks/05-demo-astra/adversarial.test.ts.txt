// Reuse only the frozen DOM fixture. Execute the frozen app with real SDK bytes.
// All network requests are in-memory mocks. Mock keys never leave this process.
const root = '/private/tmp/walleterm-audit-40d6cca9db73/';
const fixtureUrl = new URL(`file://${root}bridge/site.test.ts`);
const fixtureSource = readFileSync(fixtureUrl, 'utf8').split('const app = () =>')[0];
const helper = new Bun.Transpiler({ loader: 'ts' }).transformSync(fixtureSource)
  .replace(/^import .*\n/gm, '')
  .replaceAll('import.meta.url', JSON.stringify(fixtureUrl.href));
const contextFor = vm.runInNewContext(helper + '\ncontextFor;', {
  vm, assert, URL, URLSearchParams, AbortSignal, AbortController, browserScript,
});
const html = readFileSync(root + 'demo/site/index.html', 'utf8');
const source = browserScript(new URL(`file://${root}demo/site/app.ts`));
const key = sdk.Keypair.fromRawEd25519Seed(new Uint8Array(32).fill(61));
const destination = sdk.Keypair.fromRawEd25519Seed(new Uint8Array(32).fill(62)).publicKey();
const stranger = sdk.Keypair.fromRawEd25519Seed(new Uint8Array(32).fill(63));
const hash = (tx: sdk.Transaction) => Buffer.from(tx.hash()).toString('hex');
function transaction(amount = '0.0100000', sequence = '10') {
  return new sdk.TransactionBuilder(new sdk.Account(key.publicKey(), sequence), {
    fee: '100', networkPassphrase: sdk.Networks.TESTNET,
  }).addOperation(sdk.Operation.payment({ destination, amount, asset: sdk.Asset.native() }))
    .setTimeout(180).build();
}
function journal(state = 'signed', tx = transaction()) {
  const xdr = tx.toXDR();
  tx.sign(key);
  return { kind: 'payment', address: key.publicKey(), recipient: destination,
    hash: hash(tx), xdr, signed_xdr: tx.toXDR(), state };
}
function page(record: unknown = null, options: Record<string, any> = {}) {
  let stored = record === null ? null : JSON.stringify(record);
  let failWrites = false;
  const calls: { url: string; options: RequestInit }[] = [];
  const events: unknown[] = [];
  const storage = {
    getItem: () => stored,
    setItem: (_key: string, value: string) => {
      if (failWrites) throw Error('Injected quota failure');
      stored = value;
    },
    removeItem: () => { if (failWrites) throw Error('Injected storage failure'); stored = null; },
  };
  const fetch = async (url: string, request: RequestInit) => {
    calls.push({ url, options: request });
    if (!options.fetch) throw Error('Unexpected mock request: ' + url);
    return options.fetch(url, request);
  };
  const f = contextFor(html, { StellarSdk: sdk, crypto, Uint8Array, localStorage: storage, fetch,
    createActivityLog: () => ({ record() {}, transaction: (r: unknown) => events.push(structuredClone(r)),
      wrapFetch: (fn: unknown) => fn }), ...options.extras });
  f.run(source);
  return { ...f, calls, events, storage, getStored: () => stored,
    failWrites: (value: boolean) => { failWrites = value; } };
}
const ok = (value: unknown) => ({ ok: true, status: 200, json: async () => value });
const rejection = (status: number, value: unknown) => ({ ok: false, status, json: async () => value });

test('counterevidence: fresh signing rejects changed bytes, wrong keys, and extra signatures', async () => {
  const original = journal('review');
  const changed = transaction('0.0200000'); changed.sign(key);
  const wrong = transaction(); wrong.sign(stranger);
  const extra = transaction(); extra.sign(key, stranger);
  for (const tx of [changed, wrong, extra]) {
    const f = page(original);
    f.context.signed = tx.toXDR();
    f.run(`wallet={token:'mock',signTransaction:async()=>({signedTxXdr:signed})}; account={address:${JSON.stringify(key.publicKey())}};`);
    await f.click('sign');
    assert.equal(f.run('pending.state'), 'failed');
    assert.match(f.el('status').textContent, /failed verification/);
    assert.equal(f.calls.length, 0);
  }
});

test('finding: restored signed journal submits another valid envelope without checking its original hash', async () => {
  const record = journal();
  const other = transaction('0.0200000'); other.sign(key);
  record.signed_xdr = other.toXDR();
  const f = page(record, { fetch: async () => { throw Error('Mock response lost after POST'); } });
  assert.equal(f.run('journalBlocked'), false);
  await f.click('submit');
  assert.equal(f.calls.length, 1);
  const sent = new URLSearchParams(String(f.calls[0].options.body)).get('tx');
  assert.equal(sent, other.toXDR());
  assert.notEqual(hash(other), f.run('pending.hash'));
  assert.equal(f.run('pending.state'), 'unknown');
});

test('finding: restored unsigned envelope remains signed and can be submitted', async () => {
  const record = journal(); record.signed_xdr = record.xdr;
  const f = page(record, { fetch: async () => rejection(400, { extras: {
    result_codes: { transaction: 'tx_bad_auth' }, result_xdr: 'mock-result',
  } }) });
  assert.equal(f.run('journalBlocked'), false);
  assert.equal(f.el('submit').disabled, false);
  await f.click('submit');
  assert.equal(f.calls.length, 1);
  assert.equal(f.run('pending.state'), 'failed');
});

test('finding: malformed success JSON resolves an unknown submission as failed', async () => {
  const record = journal('unknown');
  const f = page(record, { fetch: async (url: string) => url.includes('/ledgers?')
    ? ok({ _embedded: { records: [{ closed_at: new Date().toISOString() }] } }) : ok({}) });
  await f.click('check');
  assert.equal(f.run('pending.state'), 'failed');
  assert.equal(f.run('hasFinishedTransaction()'), true);
  assert.equal(f.calls.filter((call) => call.options.method === 'POST').length, 0);
});

test('finding: a different confirmed hash unlocks the original unknown transaction', async () => {
  const record = journal('unknown');
  const f = page(record, { fetch: async (url: string) => url.includes('/ledgers?')
    ? ok({ _embedded: { records: [{ closed_at: new Date().toISOString() }] } })
    : ok({ hash: 'b'.repeat(64), ledger: 123, successful: true, result_xdr: 'mock' }) });
  await f.click('check');
  assert.equal(f.run('pending.state'), 'submitted');
  assert.notEqual(f.run('pending.result.hash'), f.run('pending.hash'));
  assert.equal(f.run('hasFinishedTransaction()'), true);
});

test('finding: the normal signed review removes the decoded operation before submission', async () => {
  const record = journal('review');
  const f = page(record);
  assert.match(f.el('details').textContent, /"amount": "0.0100000"/);
  f.context.signed = record.signed_xdr;
  f.run(`wallet={token:'mock',signTransaction:async()=>({signedTxXdr:signed})}; account={address:${JSON.stringify(key.publicKey())}};`);
  await f.click('sign');
  assert.equal(f.run('pending.state'), 'signed');
  assert.doesNotMatch(f.el('details').textContent, /"transaction"|"amount"|"time_bounds"|"fee_stroops"/);
  assert.equal(f.el('submit').hidden, false);
});

test('counterevidence: storage failure before submission sends no bytes and preserves the signed journal', async () => {
  const record = journal(); const f = page(record);
  const saved = f.getStored(); f.failWrites(true);
  await f.click('submit');
  assert.equal(f.calls.length, 0);
  assert.equal(f.getStored(), saved);
  assert.equal(f.run('busy'), false);
});

test('counterevidence: storage failure after POST leaves a durable submitting journal and no repeat POST', async () => {
  const record = journal();
  const f = page(record, { fetch: async () => {
    f.failWrites(true);
    return ok({ hash: record.hash, ledger: 123, successful: true, result_xdr: 'unused' });
  } });
  await f.click('submit');
  assert.equal(f.calls.length, 1);
  assert.equal(JSON.parse(f.getStored()).state, 'submitting');
  assert.equal(f.run('pending.state'), 'unknown');
  await f.click('submit');
  assert.equal(f.calls.length, 1);
  assert.equal(f.run('pending.state'), 'submitting');
});

test('counterevidence: malformed JSON and tx_bad_seq remain unknown; a structured rejection finishes', async () => {
  for (const [response, expected] of [
    [{ ok: true, status: 200, json: async () => { throw Error('Malformed JSON'); } }, 'unknown'],
    [rejection(504, { detail: 'timeout' }), 'unknown'],
    [rejection(400, { extras: { result_codes: { transaction: 'tx_bad_seq' }, result_xdr: 'mock' } }), 'unknown'],
    [rejection(400, { extras: { result_codes: { transaction: 'tx_bad_auth' }, result_xdr: 'mock' } }), 'failed'],
  ] as const) {
    const f = page(journal(), { fetch: async () => response });
    await f.click('submit');
    assert.equal(f.run('pending.state'), expected);
    assert.equal(f.calls.length, 1);
  }
});

test('counterevidence: IndexedDB write failure preserves in-memory history and network response', async () => {
  const history = new ActivityHistory({ store: { load: async () => [], put: async () => { throw Error('Quota'); } } });

  const response = Response.json({ hash: 'a'.repeat(64), successful: true });

test('counterevidence: normal reload submits the exact verified envelope and confirms its original hash', async () => {
  const record = journal();
  const f = page(record, { fetch: async () => ok({ hash: record.hash, ledger: 123,
    successful: true, result_xdr: 'unused-for-payment' }) });
  await f.click('submit');
  assert.equal(String(f.calls[0].options.body), new URLSearchParams({ tx: record.signed_xdr }).toString());
  assert.equal(f.run('pending.state'), 'submitted');
  assert.equal(f.run('pending.result.hash'), record.hash);
  assert.equal(f.run('pending.hash'), record.hash);
});
