const [lockPath, outputPath] = process.argv.slice(2);
if (!lockPath || !outputPath) throw new Error('Use: bun osv-query.ts LOCK OUTPUT');

const lock = await Bun.file(lockPath).text();
const packages = [...lock.matchAll(/^    "([^"]+)": \["[^"]+@([^"]+)"/gm)]
  .map((match) => ({ name: match[1], version: match[2] }))
  .filter((value, index, values) =>
    values.findIndex((candidate) => candidate.name === value.name && candidate.version === value.version) === index,
  );
const queries = packages.map(({ name, version }) => ({ package: { ecosystem: 'npm', name }, version }));
const response = await fetch('https://api.osv.dev/v1/querybatch', {
  method: 'POST',
  headers: { 'content-type': 'application/json' },
  body: JSON.stringify({ queries }),
});
if (!response.ok) throw new Error(`OSV returned ${response.status}.`);
const results = (await response.json()) as { results: { vulns?: { id: string; modified: string }[] }[] };
const evidence = {
  source: 'https://api.osv.dev/v1/querybatch',
  accessed_at: new Date().toISOString(),
  packages,
  results: packages.map((item, index) => ({ ...item, vulns: results.results[index]?.vulns ?? [] })),
};
await Bun.write(outputPath, JSON.stringify(evidence, null, 2) + '\n');
const affected = evidence.results.filter((item) => item.vulns.length > 0);
console.log(JSON.stringify({ queried: packages.length, affected }, null, 2));
