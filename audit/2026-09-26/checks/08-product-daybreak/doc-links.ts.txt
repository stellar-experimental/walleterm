import { existsSync } from 'node:fs';
import { dirname, resolve } from 'node:path';

const [root, outputPath] = process.argv.slice(2);
if (!root || !outputPath) throw new Error('Use: bun doc-links.ts ROOT OUTPUT');

const files = [
  'AGENTS.md',
  'README.md',
  'bridge/PROTOCOL.md',
  'fixtures/README.md',
  'evidence/README.md',
  ...(await Array.fromAsync(new Bun.Glob('docs/*.md').scan({ cwd: root }))),
  ...(await Array.fromAsync(new Bun.Glob('.agents/skills/**/*.md').scan({ cwd: root }))),
].sort();
const missing: { file: string; target: string }[] = [];
let localLinks = 0;
for (const file of files) {
  const text = await Bun.file(resolve(root, file)).text();
  for (const match of text.matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
    const target = match[1].trim().replace(/^<|>$/g, '');
    if (/^(?:https?:|mailto:|#)/.test(target)) continue;
    const path = decodeURIComponent(target.split('#', 1)[0]);
    if (!path) continue;
    localLinks++;
    if (!existsSync(resolve(root, dirname(file), path))) missing.push({ file, target });
  }
}
const result = {
  root,
  markdown_files: files.length,
  docs_files: files.filter((file) => file.startsWith('docs/')).length,
  skill_files: files.filter((file) => file.startsWith('.agents/skills/')).length,
  local_links: localLinks,
  missing,
};
await Bun.write(outputPath, JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify(result, null, 2));
