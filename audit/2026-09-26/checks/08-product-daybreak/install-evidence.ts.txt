import { existsSync, lstatSync, readlinkSync, realpathSync, readdirSync } from 'node:fs';
import { dirname, resolve } from 'node:path';

const [prefix, outputPath] = process.argv.slice(2);
if (!prefix || !outputPath) throw new Error('Use: bun install-evidence.ts PREFIX OUTPUT');

const command = resolve(prefix, 'bin/walleterm');
const alias = resolve(prefix, 'bin/stellar-walleterm');
const binary = realpathSync(command);
const release = resolve(dirname(binary), '..');
const manifest = await Bun.file(resolve(release, 'manifest.json')).json() as {
  files: string[];
  built_at: string;
};
const packageFiles = [
  ...(await Array.fromAsync(new Bun.Glob('node_modules/*/package.json').scan({ cwd: release }))),
  ...(await Array.fromAsync(new Bun.Glob('node_modules/@*/*/package.json').scan({ cwd: release }))),
].sort();
const packages = [];
for (const file of packageFiles) {
  const value = await Bun.file(resolve(release, file)).json() as { name?: string; version?: string };
  if (value.name && value.version) packages.push({ name: value.name, version: value.version });
}
const missingImports: { file: string; target: string }[] = [];
for await (const file of new Bun.Glob('dist/**/*.js').scan({ cwd: release })) {
  const source = await Bun.file(resolve(release, file)).text();
  for (const match of source.matchAll(/(?:from\s*|import\s*)["'](\.{1,2}\/[^"']+)["']/g)) {
    const target = resolve(release, dirname(file), match[1]);
    if (!existsSync(target)) missingImports.push({ file, target: match[1] });
  }
}
const result = {
  prefix,
  release,
  command_is_symlink: lstatSync(command).isSymbolicLink(),
  command_target: readlinkSync(command),
  alias_is_symlink: lstatSync(alias).isSymbolicLink(),
  alias_target: readlinkSync(alias),
  alias_resolves_to_binary: realpathSync(alias) === binary,
  manifest_files: manifest.files.length,
  manifest_missing: manifest.files.filter((file) => !existsSync(resolve(release, file))),
  staging_entries: readdirSync(resolve(prefix, 'share/walleterm/releases')).filter((name) => name.startsWith('.install-')),
  direct_runtime_present: ['@stellar/stellar-sdk', 'jsqr', 'qrcode'].filter((name) =>
    existsSync(resolve(release, 'node_modules', name, 'package.json')),
  ),
  direct_development_present: ['@types/bun', '@twinkleplop/bash', '@twinkleplop/json', 'prettier', 'typescript'].filter((name) =>
    existsSync(resolve(release, 'node_modules', name, 'package.json')),
  ),
  production_packages: packages,
  missing_dist_imports: missingImports,
};
await Bun.write(outputPath, JSON.stringify(result, null, 2) + '\n');
console.log(JSON.stringify({
  release,
  manifest_missing: result.manifest_missing,
  staging_entries: result.staging_entries,
  direct_runtime_present: result.direct_runtime_present,
  direct_development_present: result.direct_development_present,
  production_package_count: packages.length,
  missing_dist_imports: missingImports,
}, null, 2));
