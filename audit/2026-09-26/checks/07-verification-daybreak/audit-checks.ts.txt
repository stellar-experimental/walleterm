import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { existsSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

const root = '/private/tmp/walleterm-audit-40d6cca9db73';
const read = (path: string) => readFileSync(join(root, path), 'utf8');
const sha256 = (path: string) => createHash('sha256').update(readFileSync(join(root, path))).digest('hex');

const cliPipeline = read('tests/cli-pipeline.ts');
assert.match(cliPipeline, /stellar\(\['tx', 'send', '--network', 'testnet'\], signed\)/);
assert.doesNotMatch(cliPipeline, /ctx\.send\(/);
assert.match(cliPipeline, /ctx\.signDigest\(/);

const liveGuide = read('docs/LIVE-TESTS.md');
assert.doesNotMatch(liveGuide, /cli-pipeline/);

const submission = read('tests/submission.ts');
assert.match(submission, /fsyncSync\(fd\)/);
assert.doesNotMatch(submission, /fsyncSync\(directory\)/);
assert.match(submission, /persist\(gateFile, gate, 'wx'\)/);

const cap71 = read('tests/cap71.ts');
