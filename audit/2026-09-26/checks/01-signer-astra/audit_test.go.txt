// These tests use only disposable local mock keys and relative mock sockets.
func auditSocket(t *testing.T, serve func(net.Conn) error) (string, <-chan error) {
	t.Helper()
	dir, err := os.MkdirTemp(".", "mock-")
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.RemoveAll(dir) })
	path := filepath.Join(dir, "agent.sock")
	listener, err := net.Listen("unix", path)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = listener.Close() })
	if err := os.Chmod(path, 0600); err != nil {
		t.Fatal(err)
	}
	done := make(chan error, 1)
	go func() {
		c, err := listener.Accept()
		if err == nil {
			defer c.Close()
			_ = c.SetDeadline(time.Now().Add(4 * time.Second))
			err = serve(c)
		}
		done <- err
	}()
	return path, done
}

func TestAuditSignatureBinding(t *testing.T) {
	key, private := mockKey()
	other := ed25519.NewKeyFromSeed(bytes.Repeat([]byte{31}, 32))
	digest := make([]byte, 32)
	for i := range digest {
		digest[i] = byte(i * 7)
	}
	doubleHash := sha256.Sum256(digest)
	cases := []struct {
		name    string
		data    []byte
		private ed25519.PrivateKey
		code    int
	}{
		{"raw", digest, private, 0},
		{"hex_text", []byte(hex.EncodeToString(digest)), private, 1},
		{"base64_text", []byte(base64.StdEncoding.EncodeToString(digest)), private, 1},
		{"second_hash", doubleHash[:], private, 1},
		{"other_digest", bytes.Repeat([]byte{11}, 32), private, 1},
		{"other_key", digest, other, 1},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			path, done := auditSocket(t, func(c net.Conn) error {
				request, err := readFrame(c)
				if err != nil || !bytes.Equal(request, []byte{11}) {
					return fmt.Errorf("list request: %v", err)
				}
				if err := writeFrame(c, identities(keyBlob(key), "audit mock")); err != nil {
					return err
				}
				request, err = readFrame(c)
				if err != nil {
					return err
				}
				// Decode with a separate reader, not the production parser.
				r := bytes.NewReader(request)
				kind, err := r.ReadByte()
				if err != nil || kind != 13 {
					return errors.New("wrong request type")
				}
				var n uint32
				if err := binary.Read(r, binary.BigEndian, &n); err != nil || n != 51 {
					return errors.New("wrong key blob length")
				}
				blob := make([]byte, n)
				if _, err := io.ReadFull(r, blob); err != nil || !bytes.Equal(blob, keyBlob(key)) {
					return errors.New("wrong selected key")
				}
				if err := binary.Read(r, binary.BigEndian, &n); err != nil || n != 32 {
					return errors.New("wrong digest length")
				}
				payload := make([]byte, n)
				if _, err := io.ReadFull(r, payload); err != nil || !bytes.Equal(payload, digest) {
					return errors.New("wrong digest bytes")
				}
				if err := binary.Read(r, binary.BigEndian, &n); err != nil || n != 0 || r.Len() != 0 {
					return errors.New("flags or trailing bytes")
				}
				signed := putString(putString(nil, []byte("ssh-ed25519")), ed25519.Sign(tc.private, tc.data))
				if err := writeFrame(c, putString([]byte{14}, signed)); err != nil {
					return err
				}
				if _, err := readFrame(c); !errors.Is(err, io.EOF) {
					return fmt.Errorf("unexpected retry: %v", err)
				}
				return nil
			})
			var out, notice bytes.Buffer
			input := fmt.Sprintf(`{"public_key":%q,"digest":%q}`, encodeAddress(key), hex.EncodeToString(digest))
			code := run([]string{"sign"}, strings.NewReader(input), &out, &notice, path)
			if err := <-done; err != nil {
				t.Fatal(err)
			}
			if code != tc.code {
				t.Fatalf("exit %d: %s", code, out.String())
			}
			if code == 1 && !strings.Contains(out.String(), `"code":"invalid_signature"`) {
				t.Fatal(out.String())
			}
			if code == 0 && !strings.Contains(out.String(), `"verified":true`) {
				t.Fatal(out.String())
			}
		})
	}
}

func auditResponse(t *testing.T, body []byte, f func(net.Conn) error) error {
	t.Helper()
	client, server := net.Pipe()
	defer client.Close()
	done := make(chan struct{})
	go func() {
		defer close(done)
		defer server.Close()
		_, _ = readFrame(server)
		_ = writeFrame(server, body)
	}()
	_ = client.SetDeadline(time.Now().Add(time.Second))
	err := f(client)
	_ = client.Close()
	<-done
	return err
}

func TestAuditMalformedIdentities(t *testing.T) {
	key, _ := mockKey()
	valid := identities(keyBlob(key), "mock")
	duplicate := binary.BigEndian.AppendUint32([]byte{12}, 2)
	duplicate = append(duplicate, valid[5:]...)
	duplicate = append(duplicate, valid[5:]...)
	badKey := putString(putString(nil, []byte("ssh-ed25519")), bytes.Repeat([]byte{1}, 31))
	for name, body := range map[string][]byte{
		"duplicate":         duplicate,
		"invalid_utf8":      identities(keyBlob(key), string([]byte{0xff})),
		"short_key":         identities(badKey, "mock"),
		"key_trailing":      identities(append(keyBlob(key), 0), "mock"),
		"empty_algorithm":   identities(putString(nil, nil), "mock"),
		"overflow_length":   append([]byte{12, 0, 0, 0, 1}, 255, 255, 255, 255),
		"truncated_comment": valid[:len(valid)-1],
	} {
		t.Run(name, func(t *testing.T) {
			err := auditResponse(t, body, func(c net.Conn) error { _, err := listSigners(c); return err })
			var e *cliError
			if !errors.As(err, &e) || e.Code != "agent_protocol" {
				t.Fatalf("got %v", err)
			}
		})
	}
}

func TestAuditMalformedSignatures(t *testing.T) {
	key, private := mockKey()
	digest := bytes.Repeat([]byte{4}, 32)
	sig := ed25519.Sign(private, digest)
	wrapped := putString(putString(nil, []byte("ssh-ed25519")), sig)
	for name, body := range map[string][]byte{
		"short":            putString([]byte{14}, putString(putString(nil, []byte("ssh-ed25519")), sig[:63])),
		"long":             putString([]byte{14}, putString(putString(nil, []byte("ssh-ed25519")), append(sig, 0))),
		"wrapped_trailing": putString([]byte{14}, append(wrapped, 0)),
		"outer_trailing":   append(putString([]byte{14}, wrapped), 0),
		"failure_trailing": {5, 0},
		"unknown_type":     {6},
		"overflow_length":  {14, 255, 255, 255, 255},
	} {
		t.Run(name, func(t *testing.T) {
			err := auditResponse(t, body, func(c net.Conn) error { _, err := sign(c, keyBlob(key), digest); return err })
			var e *cliError
			if !errors.As(err, &e) || e.Code != "agent_protocol" {
				t.Fatalf("got %v", err)
			}
		})
	}
}

func TestAuditInputBounds(t *testing.T) {
	valid := fmt.Sprintf(`{"public_key":%q,"digest":%q}`, sep23Address, strings.Repeat("12", 32))
	for _, size := range []int{4095, 4096, 4097} {
		input := valid + strings.Repeat(" ", size-len(valid))
		_, _, _, err := parseInput(strings.NewReader(input))
		if (err == nil) != (size <= 4096) {
			t.Fatalf("size=%d err=%v", size, err)
		}
	}
	for _, input := range []string{`{}`, `{"digest":""}`, `{"public_key":null}`, `{"public_key":[]}`, valid + "\x00", strings.Replace(valid, "digest", "Digest", 1)} {
		if _, _, _, err := parseInput(strings.NewReader(input)); err == nil {
			t.Fatalf("accepted %q", input)
		}
	}
}

// This observation test records a deadline exception in the frozen implementation.
func TestAuditBlockedDiagnosticExceedsOperationDeadline(t *testing.T) {
	reader, writer, err := os.Pipe()
	if err != nil {
		t.Fatal(err)
	}
	defer reader.Close()
	defer writer.Close()
	if err := writer.SetWriteDeadline(time.Now().Add(50 * time.Millisecond)); err != nil {
		t.Fatal(err)
	}
	n, fillErr := writer.Write(bytes.Repeat([]byte{'x'}, 1<<20))
	if !errors.Is(fillErr, os.ErrDeadlineExceeded) || n == 0 {
		t.Fatalf("pipe fill: %d %v", n, fillErr)
	}
	if err := writer.SetWriteDeadline(time.Time{}); err != nil {
		t.Fatal(err)
	}
	key, _ := mockKey()
	listed := make(chan struct{})
	path, done := auditSocket(t, func(c net.Conn) error {
		// The mock must outlast the product deadline for this observation.
		_ = c.SetDeadline(time.Now().Add(140 * time.Second))
		if _, err := readFrame(c); err != nil {
			return err
		}
		if err := writeFrame(c, identities(keyBlob(key), "audit mock")); err != nil {
			return err
		}
		close(listed)
		if _, err := readFrame(c); !errors.Is(err, io.EOF) {
			return fmt.Errorf("unexpected signing request: %v", err)
		}
		return nil
	})
	input := fmt.Sprintf(`{"public_key":%q,"digest":%q}`, encodeAddress(key), strings.Repeat("31", 32))
	var out bytes.Buffer
	returned := make(chan int, 1)
	started := time.Now()
	go func() { returned <- run([]string{"sign"}, strings.NewReader(input), &out, writer, path) }()
	<-listed
	select {
	case code := <-returned:
		t.Fatalf("observation changed: exited early with %d", code)
	case <-time.After(122 * time.Second):
		t.Logf("Still blocked after %.3f seconds; documented deadline is 120 seconds", time.Since(started).Seconds())
	}
	_ = reader.Close()
	select {
	case code := <-returned:
		if code != 1 || !strings.Contains(out.String(), `"code":"output_error"`) {
			t.Fatalf("exit=%d output=%s", code, out.String())
		}
	case <-time.After(2 * time.Second):
		t.Fatal("process did not return after the pipe closed")
	}
	if err := <-done; err != nil {
		t.Fatal(err)
	}
	t.Log("Closing the pipe returned output_error; the mock received zero signing requests")
}

func FuzzAuditInput(f *testing.F) {
	valid := fmt.Sprintf(`{"public_key":%q,"digest":%q}`, sep23Address, strings.Repeat("12", 32))
	for _, s := range []string{valid, "{}", "null", valid + "{}", strings.Repeat("x", 4097)} {
		f.Add(s)
	}
	f.Fuzz(func(t *testing.T, input string) {
		key, digest, raw, err := parseInput(strings.NewReader(input))
		if err != nil {
			return
		}
		if len(input) > 4096 || len(raw) != 32 || hex.EncodeToString(raw) != digest {
			t.Fatal("invalid accepted input")
		}
		pub, err := decodeAddress(key)
		if err != nil || encodeAddress(pub) != key {
			t.Fatal("invalid address")
		}
		var obj map[string]string
		if err := json.Unmarshal([]byte(input), &obj); err != nil || len(obj) != 2 || obj["public_key"] != key || obj["digest"] != digest {
			t.Fatal("invalid fields")
		}
	})
}
